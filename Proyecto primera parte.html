Para el cálculo de los promedios de demora en este análisis, se tomó la decisión de filtrar la muestra para incluir únicamente los retrasos efectivos (delay > 0). Se descartaron los valores negativos (salidas o llegadas tempranas) para evitar que estos compensaran algebraicamente los retrasos. De esta forma, la métrica resultante no refleja un “balance neto” del sistema, sino la magnitud real.

query_1_sql <- "
SELECT 
    a.name AS aerolinea, 
    AVG(f.dep_delay) AS retraso_promedio
  FROM vuelos f
  JOIN aerolineas a ON f.carrier = a.carrier
  WHERE f.year = 2013 AND f.dep_delay > 0
  GROUP BY a.name
  ORDER BY retraso_promedio DESC
  LIMIT 1;
"

query_2_sql <- "
  WITH VuelosDiarios AS (
    SELECT 
      year, 
      month, 
      day,
      COUNT(*) AS cantidad_retrasos,
      -- Construimos la fecha (YYYY-MM-DD) y extraemos el número del día de la semana
      strftime('%w', year || '-' || substr('0' || month, -2, 2) || '-' || substr('0' || day, -2, 2)) AS dia_num
    FROM vuelos
    WHERE dep_delay > 0
    GROUP BY year, month, day
  )
  SELECT 
    CASE dia_num 
      WHEN '0' THEN 'Domingo'
      WHEN '1' THEN 'Lunes'
      WHEN '2' THEN 'Martes'
      WHEN '3' THEN 'Miércoles'
      WHEN '4' THEN 'Jueves'
      WHEN '5' THEN 'Viernes'
      WHEN '6' THEN 'Sábado'
    END AS dia_semana,
    AVG(cantidad_retrasos) AS promedio_retrasos
  FROM VuelosDiarios
  GROUP BY dia_num, dia_semana
  ORDER BY promedio_retrasos DESC;
"

4.- ¿Qué proporción de vuelos se retrasaron más de 30 minutos?

pregunta_4_dplyr <- vuelos %>%
  # Quitamos los vuelos cancelados (nulos) para no afectar la proporción
  filter(!is.na(dep_delay)) %>%
  summarise(
    # Contamos el total de vuelos que sí salieron
    total_vuelos = n(),
    # Sumamos 1 si se retrasó más de 30 min, y 0 si no
    vuelos_retrasados = sum(ifelse(dep_delay > 30, 1, 0), na.rm = TRUE)
  ) %>%
  # Calculamos la proporción y el porcentaje
  mutate(
    proporcion = vuelos_retrasados / total_vuelos,

  ) %>%

Como notamos que no hay ninguna columna denominada como taxi-out La calculamos como la diferencia entre la hora real de despegue y la hora programada de despegue, menos el retraso oficial de salida.

Utilizaremos el proxy taxi_out = (arr_time - dep_time) - air_time. El único detalle es que captura taxi-out + taxi-in juntos (tiempo en tierra antes de despegar y después de aterrizar), no solo el taxi-out puro. Pero es el proxy más honesto posible con las columnas disponibles.

  • Con dplyr
pregunta_12_dplyr <- vuelos %>%
  filter(
    !is.na(arr_time) & !is.na(dep_time) & !is.na(air_time)
  ) %>%
  collect() %>%
  mutate(
    # Convertimos arr_time y dep_time de HHMM a minutos desde medianoche
    arr_min  = (arr_time %/% 100) * 60 + (arr_time %% 100),
    dep_min  = (dep_time %/% 100) * 60 + (dep_time %% 100),
    # Corregimos vuelos que cruzan medianoche (arr_min < dep_min)
    arr_min  = ifelse(arr_min < dep_min, arr_min + 1440, arr_min),
    # Proxy: tiempo total de puerta a puerta menos tiempo en el aire
    taxi_total = (arr_min - dep_min) - air_time
  ) %>%
  # Filtramos anomalías: valores negativos o extremos (> 120 min)
  filter(taxi_total > 0 & taxi_total < 120) %>%
  group_by(origin) %>%
  summarise(
    taxi_total_promedio = round(mean(taxi_total, na.rm = TRUE), 2),
    total_vuelos        = n()
  ) %>%
  left_join(collect(aeropuertos), by = c("origin" = "faa")) %>%
  arrange(taxi_total_promedio) %>%
  select(name, taxi_total_promedio, total_vuelos)

# Con SQL
query_12_sql <- "
  SELECT
    a.name AS aeropuerto,
    ROUND(AVG(
      -- arr_time y dep_time de HHMM a minutos
      ((v.arr_time / 100) * 60 + (v.arr_time % 100))
      -- Ajuste por cruce de medianoche: si arr < dep sumamos 1440
      + CASE 
          WHEN ((v.arr_time / 100) * 60 + (v.arr_time % 100)) < 
               ((v.dep_time / 100) * 60 + (v.dep_time % 100))
          THEN 1440 ELSE 0 
        END
      - ((v.dep_time / 100) * 60 + (v.dep_time % 100))
      -- Restamos el tiempo en el aire
      - v.air_time
    ), 2) AS taxi_total_promedio,
    COUNT(*) AS total_vuelos
  FROM vuelos v
  JOIN aeropuertos a ON v.origin = a.faa
  WHERE
    v.arr_time  IS NOT NULL AND
    v.dep_time  IS NOT NULL AND
    v.air_time  IS NOT NULL AND
    -- Filtramos anomalías directamente en SQL
    (
      ((v.arr_time / 100) * 60 + (v.arr_time % 100))
      + CASE 
          WHEN ((v.arr_time / 100) * 60 + (v.arr_time % 100)) < 
               ((v.dep_time / 100) * 60 + (v.dep_time % 100))
          THEN 1440 ELSE 0 
        END
      - ((v.dep_time / 100) * 60 + (v.dep_time % 100))
      - v.air_time
    ) > 0
    AND
    (
      ((v.arr_time / 100) * 60 + (v.arr_time % 100))
      + CASE 
          WHEN ((v.arr_time / 100) * 60 + (v.arr_time % 100)) < 
               ((v.dep_time / 100) * 60 + (v.dep_time % 100))
          THEN 1440 ELSE 0 
        END
      - ((v.dep_time / 100) * 60 + (v.dep_time % 100))
      - v.air_time
    ) < 120
  GROUP BY v.origin, a.name
  ORDER BY taxi_total_promedio ASC;
"

20.- ¿La alta humedad se relaciona de alguna manera con los tiempos de taxi-out más largos?

Al no contar con la variable exacta de taxi-out, se utilizará la diferencia arr_time − dep_time) − air_time como una variable proxy al igual que en la pregunta 12.

  • Con dplyr
pregunta_20_dplyr <- vuelos %>%
  inner_join(clima, by = c("origin", "year", "month", "day", "hour")) %>%
  filter(
    !is.na(arr_time) & !is.na(dep_time) &
    !is.na(air_time) & !is.na(humid)
  ) %>%
  collect() %>%
  mutate(
    arr_min    = (arr_time %/% 100) * 60 + (arr_time %% 100),
    dep_min    = (dep_time %/% 100) * 60 + (dep_time %% 100),

      humid >= 80              ~ " Muy húmedo (>= 80%)"
    )
  ) %>%
  group_by(categoria_humedad) %>%
  summarise(
    taxi_total_promedio = round(mean(taxi_total, na.rm = TRUE), 2),

  ) %>%

query_20_sql <- "
  WITH taxi_calculado AS (
    SELECT
      v.origin,
      v.year, v.month, v.day, v.hour,
      -- Calculamos el proxy una sola vez en el CTE para no repetir la expresión
      ((v.arr_time / 100) * 60 + (v.arr_time % 100))
      + CASE 
          WHEN ((v.arr_time / 100) * 60 + (v.arr_time % 100)) < 
               ((v.dep_time / 100) * 60 + (v.dep_time % 100))
          THEN 1440 ELSE 0 
        END
      - ((v.dep_time / 100) * 60 + (v.dep_time % 100))
      - v.air_time AS taxi_total
    FROM vuelos v
    WHERE
      v.arr_time IS NOT NULL AND
      v.dep_time IS NOT NULL AND
      v.air_time IS NOT NULL
  )
  SELECT
    CASE
      WHEN c.humid < 40             THEN ' Seco (< 40%)'
      WHEN c.humid >= 40 AND c.humid < 60 THEN ' Normal (40 - 59%)'
      WHEN c.humid >= 60 AND c.humid < 80 THEN ' Húmedo (60 - 79%)'
      ELSE ' Muy húmedo (>= 80%)'
    END AS categoria_humedad,
    ROUND(AVG(t.taxi_total), 2) AS taxi_total_promedio,
    COUNT(*) AS total_vuelos
  FROM taxi_calculado t
  JOIN clima c ON
    t.origin = c.origin AND t.year  = c.year  AND
    t.month  = c.month  AND t.day   = c.day   AND
    t.hour   = c.hour
  WHERE
    c.humid IS NOT NULL AND
    t.taxi_total > 0    AND
    t.taxi_total < 120
  GROUP BY categoria_humedad
  ORDER BY categoria_humedad ASC;
"
